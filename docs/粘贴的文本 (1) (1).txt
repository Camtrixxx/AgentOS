def start_unitree_dds_runner(timeout=5.0):
    global DDS_PROC, DDS_CLIENT
    if DDS_PROC is not None:
        return DDS_PROC, DDS_CLIENT
    # 启动 runner 子进程（stdout 合并到主进程，便于调试）
    DDS_PROC = subprocess.Popen([sys.executable, UNITREE_RUNNER], stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    # 非阻塞地打印子进程输出线程（便于调试）
    def _drain_stdout(pipe):
        try:
            for line in pipe:
                print("[unitree_runner]", line.rstrip())
        except Exception:
            pass
    threading.Thread(target=_drain_stdout, args=(DDS_PROC.stdout,), daemon=True).start()

    # 尝试连接 runner 的 TCP 服务
    start = time.time()
    client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    while True:
        try:
            client.connect((DDS_HOST, DDS_PORT))
            client.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
            DDS_CLIENT = client
            print("info: connected to unitree_dds_runner tcp server")
            return DDS_PROC, DDS_CLIENT
        except Exception:
            if time.time() - start > timeout:
                # 若不能连接，停止子进程并抛异常
                try:
                    DDS_PROC.terminate()
                except Exception:
                    pass
                DDS_PROC = None
                client.close()
                raise RuntimeError("failed to connect to unitree_dds_runner tcp server")
            time.sleep(0.1)

def stop_unitree_runner():
    global DDS_PROC, DDS_CLIENT
    try:
        if DDS_CLIENT:
            DDS_CLIENT.close()
    except Exception:
        pass
    try:
        if DDS_PROC:
            DDS_PROC.terminate()
            DDS_PROC.wait(timeout=2)
    except Exception:
        try:
            DDS_PROC.kill()
        except Exception:
            pass
    DDS_PROC = None
    DDS_CLIENT = None

class ReceiveSignal:
    """接收信号状态，用于在 receiver 和 rl_updater 之间传递接收到的信号数据"""
    def __init__(self):
        self.lock = threading.Lock()
        self.signals = {}  # 存储所有接收到的信号数据
        self.signal_timestamps = {}  # 存储每个信号的更新时间戳

    def set_signal(self, signal_name: str, signal_data: dict):
        """设置信号数据"""
        with self.lock:
            self.signals[signal_name] = signal_data
            self.signal_timestamps[signal_name] = time.time()

    def set_signals(self, signals: dict):
        """批量设置多个信号数据"""
        with self.lock:
            current_time = time.time()
            for signal_name, signal_data in signals.items():
                self.signals[signal_name] = signal_data
                self.signal_timestamps[signal_name] = current_time

    def get_signal(self, signal_name: str, timeout: float = 5.0):
        """
        获取指定信号数据（带超时检查）

        Args:
            signal_name: 信号名称
            timeout: 超时时间（秒）

        Returns:
            dict: 信号数据，如果超时或不存在则返回 None
        """
        with self.lock:
            if signal_name not in self.signals:
                return None

            timestamp = self.signal_timestamps.get(signal_name, 0)
            data_age = time.time() - timestamp

            if data_age > timeout:
                return None

            return self.signals[signal_name].copy() if isinstance(self.signals[signal_name], dict) else self.signals[signal_name]

    def get_all_signals(self, timeout: float = 5.0):
        """
        获取所有信号数据（带超时检查）

        Args:
            timeout: 超时时间（秒）

        Returns:
            dict: 所有未超时的信号数据
        """
        with self.lock:
            result = {}
            current_time = time.time()

            for signal_name, signal_data in self.signals.items():
                timestamp = self.signal_timestamps.get(signal_name, 0)
                data_age = current_time - timestamp

                if data_age <= timeout:
                    result[signal_name] = signal_data.copy() if isinstance(signal_data, dict) else {"data": signal_data}
                    result[signal_name]["timestamp"] = timestamp

            return result

    def has_signal(self, signal_name: str, timeout: float = 5.0):
        """检查信号是否存在且未超时"""
        with self.lock:
            if signal_name not in self.signals:
                return False

            timestamp = self.signal_timestamps.get(signal_name, 0)
            data_age = time.time() - timestamp

            return data_age <= timeout


def publish_cmds(frame_data: np.ndarray, frame_idx: int, sim_joint_limits, pub):
    pose_start_idx = 8
    valid_hand_idx = [2, 6, 4, 0, 9, 8]
    left_hand_start_idx = 30
    right_hand_start_idx = 49
    hand_dof = 6 * 2

    hand_angles = [frame_data[left_hand_start_idx + idx] for idx in valid_hand_idx ] + \
        [frame_data[right_hand_start_idx + idx] for idx in valid_hand_idx ]
    
    hand_values = []
    for i, angle in enumerate(hand_angles):
        real_value = 1000 * (1 - angle / sim_joint_limits[i])
        # 确保数值在0-1000范围内
        real_value = max(0, min(1000, real_value))
        hand_values.append(int(real_value)) 
    hand_set = hand_values
    # print(f"hand_set: {hand_set}, time:{time.time()}")

    pose_set = frame_data[pose_start_idx:left_hand_start_idx] + \
        frame_data[left_hand_start_idx + hand_dof : right_hand_start_idx]
    if frame_idx % 100 == 0:
        print(f"[{frame_idx}]pub: hand_set: {hand_set}, pose_set: {pose_set}")

    hand_mode = 0b0001
    pose_mode = 0b0000
    # 如果主进程内有本地 dds pub（不建议），直接写；否则通过 TCP 发送到 runner
    if pub is not None:
        try:
            # 在少数情况下主进程仍可能持有 full_body_ctrl binding
            cmd_msg = full_body_ctrl(hand_mode=hand_mode, pose_mode=pose_mode, hand_set=hand_set, pose_set=pose_set)
            pub.Write(cmd_msg, 0.1)
        except Exception as e:
            print("warning: local DDS publish failed:", e)
    else:
        # 通过 TCP 发送给 runner：hand_mode, pose_mode, hand_set, pose_set
        try:
            if DDS_CLIENT is not None:
                payload = {"hand_mode": hand_mode, "pose_mode": pose_mode, "hand_set": hand_set, "pose_set": list(pose_set)}
                payload = {
                        "hand_mode": int(hand_mode),
                        "pose_mode": int(pose_mode),
                        "hand_set": [int(x) for x in hand_set],
                        "pose_set": [float(x) for x in pose_set],
                    }
                try:
                    DDS_CLIENT.sendall((json.dumps(payload) + "\n").encode())
                except Exception as e:
                    print("warning: failed to send cmd to unitree runner:", e)
            else:
                # no dds client available
                pass
        except Exception as e:
            print("warning: error preparing dds payload:", e)


def reorder_signals(signals: dict):
    """
    根据信号名称重新排序信号
    """
    qpos = [signals["lowstate"]["motor_state"][i]["q"] for i in range(29)]
    qvel = [signals["lowstate"]["motor_state"][i]["dq"] for i in range(29)]
    quat_pelvis = list(signals["lowstate"]["imu"]["quaternion"])
    gyro_pelvis = list(signals["lowstate"]["imu"]["gyroscope"])
    gvec_pelvis = list(signals["lowstate"]["imu"]["accelerometer"])
    timestamp = signals["lowstate"]["timestamp"]

    return qpos + qvel + gyro_pelvis + quat_pelvis + [timestamp]

def signal_receiver_thread(signal_receiver: SignalReceiver, receive_signal: ReceiveSignal, stop_evt: threading.Event):
    """
    在单独线程中运行信号接收器，并将接收到的信号更新到接收信号状态

    Args:
        signal_receiver: SignalReceiver 节点实例
        receive_signal: 接收信号状态
        stop_evt: 停止事件
    """
    executor = rclpy.executors.SingleThreadedExecutor()
    executor.add_node(signal_receiver)

    update_interval = 0.01  # 每 0.01 秒更新一次接收信号状态
    last_update_time = time.time()

    try:
        while not stop_evt.is_set():
            executor.spin_once(timeout_sec=0.1)

            # 定期从 signal_receiver 获取所有信号并更新到接收信号状态
            current_time = time.time()
            if current_time - last_update_time >= update_interval:
                try:
                    # 获取所有接收到的信号
                    all_signals = signal_receiver.get_all_signals(timeout=0.1)
                    if all_signals:
                        # 提取信号数据并更新到接收信号状态
                        signals_dict = {}
                        for signal_name, signal_info in all_signals.items():
                            signals_dict[signal_name] = signal_info['data']
                        receive_signal.set_signals(signals_dict)
                    last_update_time = current_time
                except Exception as e:
                    signal_receiver.get_logger().error(f'Error updating receive signal state: {e}')
    except Exception as e:
        signal_receiver.get_logger().error(f'Error in signal_receiver_thread: {e}')
    finally:
        executor.remove_node(signal_receiver)


def print_received_signals(signal_receiver: SignalReceiver, print_interval: int = 100):
    """
    定期打印接收到的信号统计信息

    Args:
        signal_receiver: SignalReceiver 节点实例
        print_interval: 打印间隔（消息数）
    """
    stats = signal_receiver.get_receive_stats()
    if stats:
        print("\n=== 信号接收统计 ===")
        for topic, count in sorted(stats.items()):
            print(f"{topic}: {count} 条消息")
        print("=" * 30 + "\n")


def main(args=None):
    rl_fps = 50.0
    # 启动 unitree DDS runner 子进程（避免在本进程初始化 CycloneDDS）
    try:
        start_unitree_dds_runner()
    except Exception as e:
        print("warning: failed to start unitree runner:", e)

    rclpy.init(args=args)

    # 不在本进程初始化 dds 句柄；将 dds_publisher/ds_subscriber 置为 None，publish_cmds 会通过 socket 转发
    dds_publisher = None
    dds_subscriber = None
    
    publish_signal = PublishSignal()  # 创建发布信号状态（从 rl_node 发送出去的消息）
    receive_signal = ReceiveSignal()  # 创建接收信号状态（从 joint_receiver 接收的消息）
    stop_evt = threading.Event()

    node = JointPublisher(rl_fps, publish_signal, dds_pub=dds_publisher, dds_sub = dds_subscriber)

    # 创建信号接收器
    signal_receiver = SignalReceiver()

    # 启动信号接收器线程，传入接收信号状态
    receiver_thread = threading.Thread(
        target=signal_receiver_thread,
        args=(signal_receiver, receive_signal, stop_evt),
        daemon=True,
    )
    receiver_thread.start()

    # 启动 RL 更新线程，传入发布信号状态和接收信号状态
    updater_thread = threading.Thread(
        target=rl_updater,
        args=(rl_fps, publish_signal, receive_signal, stop_evt),
        daemon=True,
    )
    updater_thread.start()

    try:
        rclpy.spin(node)          # 单线程执行器，回调与主线程同线程运行
    except KeyboardInterrupt:
        pass
    finally:
        stop_evt.set()
        updater_thread.join()
        receiver_thread.join()
        # 打印接收统计
        print_received_signals(signal_receiver)
        signal_receiver.destroy_node()
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
